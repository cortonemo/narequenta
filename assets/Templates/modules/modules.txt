    // Function to handle the Python-driven sheet switching
    function showSheet(targetKey) {
		if (window.pywebview && window.pywebview.api && window.pywebview.api.load_sheet) {
			window.pywebview.api.load_sheet(targetKey).then(html => {
				document.open();
				document.write(html);
				document.close();
			});
} else {
			alert(`Simulation: Requesting sheet assembly for: ${targetKey}. pywebview API not available.`);
}
	}
